# ================= OPTIMIZATION NOTES =================
# - Ranger filter is rebuilt at most once per slow tick (was up to 5x).
# - Hull probe runs at 10 Hz with a cached wall normal; collision response stays 20 Hz.
# - Yaw rotations are reused as basis vectors instead of recomputed per target.
# - Aim point / stow angles / shoulder stow quaternion computed once per tick.
# - Redundant holoAng, toLocal and sound API calls removed.
# - Control flow uses the canonical `} elseif` / `} else` form from the Wiremod wiki.
# Tuning values and visible behaviour are unchanged.

# ================= FUNCTIONS =================
function angle slewAng(Cur:angle, Tar:angle, Rate, DD) {
    local Step = Rate*DD
    local DP = angnorm(Tar:pitch() - Cur:pitch())
    local DY = angnorm(Tar:yaw() - Cur:yaw())
    local DR = angnorm(Tar:roll() - Cur:roll())
    if (abs(DP) < 0.2) { DP = 0 }
    if (abs(DY) < 0.2) { DY = 0 }
    if (abs(DR) < 0.2) { DR = 0 }
    return ang(Cur:pitch() + clamp(DP, -Step, Step), Cur:yaw() + clamp(DY, -Step, Step), Cur:roll() + clamp(DR, -Step, Step))
}
function buildRangerFilter() {
    RigFilter:clear()
    local Roots = array(CurBase, Seat, Torso, Pelvis, LHip, LThigh, LFoot, RHip, RThigh, RFoot, Gun_Right, Gun_Left, Gun_Right_Ent, Gun_Left_Ent, Shoulder_Base_1, Shoulder_Base_2, Shoulder_Base_3, Shoulder_Base_4, Ignore1, Ignore2, Ignore3, Ignore4)
    foreach (I, E:entity = Roots) {
        if (E) {
            RigFilter:pushEntity(E)
            foreach (J, C:entity = E:children()) {
                if (C) {
                    RigFilter:pushEntity(C)
                    foreach (K, G:entity = C:children()) {
                        if (G) {
                            RigFilter:pushEntity(G)
                            foreach (L, H2:entity = G:children()) { if (H2) { RigFilter:pushEntity(H2) } }
                        }
                    }
                }
            }
        }
    }
    for (I = 1, 21) {
        local H = holoEntity(I)
        if (H) { RigFilter:pushEntity(H) }
    }
    foreach (I, E:entity = CurBase:getConnectedEntities()) { if (E) { RigFilter:pushEntity(E) } }
    foreach (I, E:entity = players()) { if (E) { RigFilter:pushEntity(E) } }
    rangerFilter(RigFilter)
}
function angle shoulderStowAng() {
    return CurBase:toWorld(ang(Shoulder_Stow_Pitch, Shoulder_Stow_Yaw, Shoulder_Stow_Roll))
}
function number setupShoulder(Index, ShoulderBase:entity) {
    if (!ShoulderBase | !CurBase) { return 0 }
    holoCreate(Index, ShoulderBase:toWorld(vec(0,0,0)), vec(0), ShoulderBase:toWorld(ang(0,0,0)), vec(255), "cube")
    local ShoulderChildren = ShoulderBase:children()
    CamFilter:pushEntity(ShoulderBase)
    CamFilter:pushEntity(holoEntity(Index))
    local ShoulderLocalPos = CurBase:toLocal(ShoulderBase:pos())
    local Y = ShoulderLocalPos:y()
    local Side = 2
    if (abs(Y) < 1) {
        if ((Index % 2) == 1) { Side = 1 } else { Side = 2 }
    } elseif (Y >= 0) { Side = 1 }
    if (Index == 13) { ShoulderSide1 = Side }
    if (Index == 14) { ShoulderSide2 = Side }
    if (Index == 15) { ShoulderSide3 = Side }
    if (Index == 16) { ShoulderSide4 = Side }
    ShoulderBase:parentTo(holoEntity(Index))
    holoPos(Index, CurBase:toWorld(ShoulderLocalPos))
    holoAng(Index, shoulderStowAng())
    if (TorsoReady) { holoParent(Index, 21) } else { holoParent(Index, CurBase) }
    holoEntity(Index):propFreeze(1)
    ShoulderBase:propFreeze(1)
    for (I = 1, ShoulderChildren:count()) { ShoulderChildren[I, entity]:parentTo(ShoulderBase) }
    FilterDirty = 1
    return 1
}
function stowShoulders() {
    if (CurBase) {
        local StowAng = shoulderStowAng()
        for (I = 1, 4) {
            local Index = 12 + I
            if (holoEntity(Index)) { holoAng(Index, StowAng) }
        }
    }
}
function two_seg_ik(IA, IB, IK, A:entity, B:entity, P:entity, Target:vector) {
    local AxisLocal = P:toLocalAxis(Target - A:pos())
    holoAng(IA, P:toWorld(ang(0, 0, atan(AxisLocal:z(), AxisLocal:y()) + 90)))
    local AxisLocal2 = A:toLocalAxis(Target - B:pos())
    local Distance = AxisLocal2:length()
    if (Distance > Length_AB) { Distance = Length_AB }
    if (Distance < 1) { Distance = 1 }
    local D2 = Distance*Distance
    local HipArg = clamp((D2 + Length_A2 - Length_B2) / (2*Distance*Length_A), -1, 1)
    local KneeArg = clamp((Length_B2 + Length_A2 - D2) / Length_BA2, -1, 1)
    holoAng(IB, A:toWorld(ang(atan(-AxisLocal2:x(), AxisLocal2:z()) + acos(HipArg), 180, 0)))
    holoAng(IK, B:toWorld(ang(acos(KneeArg) + 180, 0, 0)))
}

            holoAng(99, CurBase:angles())
            RebasePhase = 1
        }
    } elseif (RebasePhase == 1) {
        PendingBase:parentTo()
        holoDelete(99)
        PendingBase:propSetVelocity(vec(0))
        PendingBase:propFreeze(1)
        holoUnparent(1)
        holoParent(1, PendingBase)
        local OldBase = CurBase
        CurBase = PendingBase
        Mass = CurBase:mass()
        CurBase:propInertia(vec(3500,3500,3500))
        CurBase:propGravity(!On_Ground)
        if (Seat:driver():isPlayer()) { CurBase:propFreeze(0) }
        OldBase:propFreeze(1)
        CamFilter:clear()
        CamFilter:pushEntity(Gun_Right_Ent)
        CamFilter:pushEntity(Gun_Left_Ent)
        foreach (I, E:entity = CurBase:getConstraints()) { CamFilter:pushEntity(E) }
        foreach (I, E:entity = CurBase:children()) { CamFilter:pushEntity(E) }
        FilterDirty = 1
        MaxFallZ = CurBase:pos():z()
        ChassisYaw = CurBase:angles():yaw()
        YawCustom = ChassisYaw
        Idle = 0
        CamDirty = 1
        DoSlow = 1
        DoGround = 1
        RebasePhase = 0
        PendingBase = noentity()
    }

    local Drv = Seat:driver()
    KW = clamp((Drv:keyForward() | Drv:keyPressed("w") | Drv:keyPressed("W")) + Debug, 0, 1)
    KA = clamp(Drv:keyLeft() | Drv:keyPressed("a") | Drv:keyPressed("A"), 0, 1)
    KS = clamp(Drv:keyBack() | Drv:keyPressed("s") | Drv:keyPressed("S"), 0, 1)
    KD = clamp(Drv:keyRight() | Drv:keyPressed("d") | Drv:keyPressed("D"), 0, 1)
    Mouse1 = Drv:keyAttack1()
    Mouse2 = Drv:keyAttack2()
    Alt = Drv:keyWalk()
    XKey = Drv:keyPressed("x")
    BKey = Drv:keyPressed("b")
    Active = (Drv:isPlayer() + Debug) * PartsOK

    if (DoSlow) {
        PartsOK = CurBase & Pelvis & Seat & RHip & LHip & LFoot & RFoot & LThigh & RThigh
        if (changed(Ignore1) | changed(Ignore2) | changed(Ignore3) | changed(Ignore4)) { FilterDirty = 1 }
        if (!ShoulderHoloReady1 & Shoulder_Base_1 & CurBase) { ShoulderHoloReady1 = setupShoulder(13, Shoulder_Base_1) }
        if (!ShoulderHoloReady2 & Shoulder_Base_2 & CurBase) { ShoulderHoloReady2 = setupShoulder(14, Shoulder_Base_2) }
        if (!ShoulderHoloReady3 & Shoulder_Base_3 & CurBase) { ShoulderHoloReady3 = setupShoulder(15, Shoulder_Base_3) }
        if (!ShoulderHoloReady4 & Shoulder_Base_4 & CurBase) { ShoulderHoloReady4 = setupShoulder(16, Shoulder_Base_4) }
    }
    if (FilterDirty) { buildRangerFilter() FilterDirty = 0 }

    if (changed(BKey) & BKey & Active) { CameraSide = -CameraSide CamDirty = 1 }
    if (changed(Alt) & Alt) { ThirdPerson = !ThirdPerson CamDirty = 1 CamDist = TP_Dist*ThirdPerson }
    if (changed(ZMState) | changed(ThirdPerson)) { CamFOV = FOV[clamp(ZMState+1, 1, FOV:count()), number] }
    if (TP_Dist != TP_DistTarget) {
        TP_Dist = TP_Dist + (TP_DistTarget - TP_Dist)*TP_DistLerp
        if (abs(TP_DistTarget - TP_Dist) < 0.5) { TP_Dist = TP_DistTarget }
        CamDist = TP_Dist*ThirdPerson
    }

    local BasePos = CurBase:pos()
    local Vel = CurBase:vel()
    local HasMoveInput = (KW | KA | KS | KD)

    if (DoGround) {
        if (GroundWorldOnly) { rangerHitEntities(0) }
        local GR = rangerOffsetHull(RangerDist, BasePos, -ZAxis, RangerHull)
        if (GroundWorldOnly) { rangerHitEntities(1) }
        Dist = GR:distance() + Offset:z()
        GNrm = GR:hitNormal()
        GHit = GR:pos()
        GVel = GR:entity():vel()*GroundVelScale
        GWorld = !GR:entity()
        local NewOG = GR:hit() & (Dist < StandHeight + 25)
        if (NewOG != On_Ground) {
            On_Ground = NewOG
            CurBase:propGravity(!On_Ground)
            if (Idle) { Idle = 0 CurBase:propFreeze(0) }
            if (On_Ground) {
                local FallDistance = MaxFallZ - BasePos:z()
                if (((FallDistance > FallFXHeight) | (-Vel:z() > FallFXSpeed)) & (Curtime > (LastLandTime + LandingFXCooldown))) {
                    LastLandTime = Curtime
                    CurBase:emitSound("ambient/machines/thumper_hit.wav", 100, 100)
                }
            }
            MaxFallZ = BasePos:z()
        }
    }
    if (!On_Ground) { if (BasePos:z() > MaxFallZ) { MaxFallZ = BasePos:z() } }

    if (Idle) {
        if (HasMoveInput | !On_Ground | (abs(StandHeight - Dist) > 4)) {
            Idle = 0
            CurBase:propFreeze(0)
            CurBase:propGravity(!On_Ground)
            CamDirty = 1
        }
    } elseif (IdleLock & !HasMoveInput & On_Ground) {
        local AV = CurBase:angVel()
        local Spin = abs(AV:pitch()) + abs(AV:yaw()) + abs(AV:roll())
        if ((Vel:length() < 0.5) & (Spin < 0.5) & (abs(StandHeight - Dist) < 1.5)) {
            Idle = 1
            CurBase:propSetVelocity(vec(0))
            CurBase:propFreeze(1)
        }
    }

    AimRight = clamp(AimRight + GunDeploySpeed*(Mouse2 - !Mouse2)*Dt, Edit, 1)
    AimLeft = clamp(AimLeft + GunDeploySpeed*(Mouse1 - !Mouse1)*Dt, Edit, 1)
    FireLeft = (AimLeft >= 1) & !Edit & FireEnabled
    FireRight = (AimRight >= 1) & !Edit & FireEnabled

    W = KW & Active
    S = KS & Active
    A = KA & Active
    D = KD & Active
    AimMode = 1

    if (!Idle) {
        YawCustom = angnorm(YawCustom + AimTurnSpeed*(KA - KD)*Dt)
        local YawErr = angnorm(YawCustom - ChassisYaw)
        if (abs(YawErr) < 0.2) { YawErr = 0 }
        local MaxTurn = ChassisTurnRate*Dt
        ChassisYaw = angnorm(ChassisYaw + clamp(YawErr, -MaxTurn, MaxTurn))

        local Nrm = ZAxis
        if (GWorld | !TerrainWorldOnly) {
            if (GNrm:length() > 0.5) { Nrm = GNrm }
        }
        if (Nrm:z() < 0) { Nrm = -Nrm }
        if (!On_Ground) { Nrm = ZAxis }
        local MinNZ = cos(TerrainMaxTilt)
        if (Nrm:z() < MinNZ) {
            local Hv = vec(Nrm:x(), Nrm:y(), 0)
            local HL = Hv:length()
            if (HL > 0.0001) { Nrm = (Hv*(sin(TerrainMaxTilt)/HL) + vec(0,0,MinNZ)):normalized() } else { Nrm = ZAxis }
        }
        # Rotation basis is reused by terrain projection, hull probe, foot slots and foot normals.
        local MoveF = vec(1,0,0):rotateAroundAxis(ZAxis, ChassisYaw)
        local MoveR = vec(0,1,0):rotateAroundAxis(ZAxis, ChassisYaw)
        local TLerp = On_Ground ? TerrainLerp : TerrainAirLerp
        TerrainNormal = (TerrainNormal + (Nrm - TerrainNormal)*TLerp):normalized()
        local TNx = TerrainNormal:dot(MoveF)
        local TNy = TerrainNormal:dot(MoveR)
        local TNz = TerrainNormal:z()
        TerrainPitch = clamp(atan(TNx, TNz), -TerrainMaxTilt, TerrainMaxTilt)
        TerrainRoll = clamp(-atan(TNy, TNz), -TerrainMaxTilt, TerrainMaxTilt)

        local MoveVec = MoveF*(KW - KS)
        local VelH = vec(Vel:x(), Vel:y(), 0)
        local SpeedH = VelH:length()
        local MoveLen = MoveVec:length()
        local NeedProbe = HullEnable & ((SpeedH > 2) | (MoveLen > 0.01))
        local DoHull = (GndN == 0) | (GndN == 2)
        local PD = vec(0)
        local HaveProbe = 0
        if (!NeedProbe) {
            HullHitPos = vec(0)
            HullHitN = vec(0)
            HullHitUntil = 0
        } elseif (MoveLen > 0.01) {
            local ProbeDir = VelH + (MoveVec/MoveLen)*60
            local ProbeLen = ProbeDir:length()
            if (ProbeLen > 1) { PD = ProbeDir/ProbeLen HaveProbe = 1 }
        } elseif (SpeedH > 1) {
            PD = VelH/SpeedH
            HaveProbe = 1
        }
        if (DoHull & HaveProbe) {
            local HR = -MoveR
            local HX = abs(MoveF:x())*HullLen + abs(HR:x())*HullWid
            local HY = abs(MoveF:y())*HullLen + abs(HR:y())*HullWid
            local HullCenter = BasePos + MoveF*HullFwd + ZAxis*HullLift
            local Look = HullLook + SpeedH*0.15
            rangerHitEntities(1)
            local RH = rangerOffsetHull(Look, HullCenter, PD, vec(HX,HY,HullH))
            if (RH:hit()) {
                HullHitPos = RH:pos()
                HullHitN = RH:hitNormal()
                HullHitUntil = (RH:distance() > 1) ? Curtime + 0.12 : 0
            } else {
                HullHitPos = vec(0)
                HullHitN = vec(0)
                HullHitUntil = 0
            }
        }
        # Wall response is applied on every fast tick using the cached normal.
        if (HaveProbe & (Curtime < HullHitUntil)) {
            local N = HullHitN
            if (abs(N:z()) < 0.6) {
                local NH = vec(N:x(), N:y(), 0)
                local NHL = NH:length()
                if (NHL > 0.01) {
                    local WN = NH/NHL
                    if (WN:dot(PD) < -0.2) {
                        local DriveInto = MoveVec:dot(WN)
                        if (DriveInto < 0) { MoveVec = MoveVec - WN*DriveInto }
                        local VelInto = VelH:dot(WN)
                        if (VelInto < 0) { CurBase:propSetVelocity(Vel - WN*VelInto*HullBrake) }
                    }
                }
            }
        }
        local MoveLen2 = MoveVec:length()
        local MoveMag = clamp(MoveLen2, 0, 1)
        local MoveDir = (MoveLen2 > 0) ? MoveVec/MoveLen2 : vec(0)

        if (Active & HasMoveInput) {
            if (CurrentMoveSpeed < Move_Speed_Min) { CurrentMoveSpeed = Move_Speed_Min }
            CurrentMoveSpeed = clamp(CurrentMoveSpeed + Move_Accel*Dt, Move_Speed_Min, Move_Speed)
        } else {
            CurrentMoveSpeed = clamp(CurrentMoveSpeed - Move_Decel*Dt, 0, Move_Speed)
        }

        holoAng(1, ang(TerrainPitch, ChassisYaw, TerrainRoll))
        # Torso angle is already set by the visual tick, no need to duplicate it here.

        local HErr = StandHeight - Dist
        if (abs(HErr) < 1) { HErr = 0 }
        local Stand_Force = vec(0,0,HErr*5)*On_Ground
        local DampXY = HasMoveInput ? Move_Damping : Coast_Damping
        local DampZ = 0.1 + 0.8*On_Ground
        local Damping_Force = -Vel*vec(DampXY, DampXY, DampZ)
        local Move_Force = MoveDir*CurrentMoveSpeed*MoveMag
        local AE = CurBase:toLocal(ang(TerrainPitch, ChassisYaw, TerrainRoll))
        CurBase:applyForce((GVel + Stand_Force + Damping_Force + Move_Force)*Mass)
        CurBase:applyAngForce((ang(angnorm(AE:pitch()), angnorm(AE:yaw()), angnorm(AE:roll()))*45 - CurBase:angVel()*10)*Mass)

        # ---- :    ,   ----
        local StanceProj = Leg_Stand_Width*cos(TerrainRoll)
        local FwdProj = Leg_Forward_Offset*cos(TerrainPitch)
        local LSlot = BasePos + MoveF*FwdProj + MoveR*StanceProj
        local RSlot = BasePos + MoveF*FwdProj - MoveR*StanceProj
        local NomL = HP:toWorld(LHeight)
        local NomR = HP:toWorld(RHeight)
        local LTarget = NomL
        local RTarget = NomR
        local FootUpL = TerrainNormal
        local FootUpR = TerrainNormal
        if (On_Ground & !Edit & (GNrm:z() > 0.2)) {
            local NZ = GNrm:z()
            local GZL = GHit:z() - (GNrm:x()*(LSlot:x()-GHit:x()) + GNrm:y()*(LSlot:y()-GHit:y()))/NZ + FootGroundOffset
            local GZR = GHit:z() - (GNrm:x()*(RSlot:x()-GHit:x()) + GNrm:y()*(RSlot:y()-GHit:y()))/NZ + FootGroundOffset
            GZL = clamp(GZL, NomL:z() - FootTravelDown, NomL:z() + FootTravelUp)
            GZR = clamp(GZR, NomR:z() - FootTravelDown, NomR:z() + FootTravelUp)
            LTarget = vec(LSlot:x(), LSlot:y(), GZL)
            RTarget = vec(RSlot:x(), RSlot:y(), GZR)
            FootUpL = GNrm
            FootUpR = GNrm
        }
        two_seg_ik(2, 3, 4, HLA, HLH, HP, LTarget)
        two_seg_ik(7, 8, 9, HRA, HRH, HP, RTarget)
        local NFx = FootUpL:dot(MoveF)
        local NFy = FootUpL:dot(MoveR)
        local NFz = FootUpL:z()
        holoAng(5, ang(clamp(atan(NFx, NFz), -50, 50), ChassisYaw + FootSplay, clamp(-atan(NFy, NFz), -50, 50)))
        local NRx = FootUpR:dot(MoveF)
        local NRy = FootUpR:dot(MoveR)
        local NRz = FootUpR:z()
        holoAng(10, ang(clamp(atan(NRx, NRz), -50, 50), ChassisYaw - FootSplay, clamp(-atan(NRy, NRz), -50, 50)))

        CamDirty = 1
    }

    local ShoulderAnyReady = ShoulderHoloReady1 | ShoulderHoloReady2 | ShoulderHoloReady3 | ShoulderHoloReady4
    if (changed(XKey) & XKey & Active & ShoulderAnyReady) {
        ShoulderMode = !ShoulderMode
        if (ShoulderMode) {
            ShoulderDeploy = 1
            ShoulderDeployTime = Curtime + ShoulderDeployDelay
            ShoulderStowUntil = 0
        } else {
            ShoulderDeploy = 0
            ShoulderStowUntil = Curtime + Shoulder_Stow_Time
        }
    }
    if ((ShoulderMode & ShoulderDeploy) & (Curtime >= ShoulderDeployTime)) { ShoulderDeploy = 0 }
    if (!Active | !ShoulderAnyReady) {
        if (ShoulderMode | (ShoulderStowUntil > 0)) { stowShoulders() }
        ShoulderMode = 0
        ShoulderDeploy = 0
        ShoulderStowUntil = 0
    }
    if (ShoulderMode & ShoulderAnyReady & !ShoulderDeploy & FireEnabled & !Edit) {
        FireShoulderLeft = Mouse1
        FireShoulderRight = Mouse2
    } else {
        FireShoulderLeft = 0
        FireShoulderRight = 0
    }
    FireShoulder = FireShoulderLeft | FireShoulderRight
    FireShoulder1 = ShoulderHoloReady1*((ShoulderSide1 == 1) ? FireShoulderLeft : FireShoulderRight)
    FireShoulder2 = ShoulderHoloReady2*((ShoulderSide2 == 1) ? FireShoulderLeft : FireShoulderRight)
    FireShoulder3 = ShoulderHoloReady3*((ShoulderSide3 == 1) ? FireShoulderLeft : FireShoulderRight)
    FireShoulder4 = ShoulderHoloReady4*((ShoulderSide4 == 1) ? FireShoulderLeft : FireShoulderRight)

    if ((Active & ShoulderMode) | (ShoulderStowUntil > Curtime)) {
        if (Active & ShoulderMode & !ShoulderDeploy) {
            local STar = ang(clamp(CamPitchS + Shoulder_Aim_Pitch_Offset, -80, 80), CamYawS + Shoulder_Aim_Yaw_Offset, Shoulder_Aim_Roll_Offset)
            for (I = 1, 4) {
                local Index = 12 + I
                local S = holoEntity(Index)
                if (S) { holoAng(Index, slewAng(S:angles(), STar, TorsoTurnRate, Dt)) }
            }
        } else {
            local StowQ = quat(shoulderStowAng())
            local T = clamp(Shoulder_Stow_Speed/100, 0.01, 1)
            for (I = 1, 4) {
                local Index = 12 + I
                local S = holoEntity(Index)
                if (S) { holoAng(Index, slerp(quat(S:angles()), StowQ, T):toAngle()) }
            }
        }
    }
    if (!ShoulderMode & ShoulderStowUntil > 0 & Curtime >= ShoulderStowUntil) { stowShoulders() ShoulderStowUntil = 0 }

    if (!ThirdPerson) { if (ZMState < 0) { ZMState = 0 } }

    if (!Idle | (EngineCurrentVol > 0) | (SlideVol > 0)) {
        local SpeedRef = max(Move_Speed, 1)
        local GroundSpeed2 = vec(Vel:x(), Vel:y(), 0):length()
        local BoostAny = (W | A | S | D) & Active & !Edit
        if (BoostAny & !BoostWasOn) {
            soundStop(30)
            CurBase:soundPlay(30, 0, "Phx.Afterburner3")
            soundPitch(30, 45)
            soundVolume(30, 0)
            BoostSoundPlaying = 1
            EngineVolLast = -1
        }
        local BoostTarget = BoostAny ? EngineVolume : 0
        EngineCurrentVol = EngineCurrentVol + (BoostTarget - EngineCurrentVol)*0.25
        if (EngineCurrentVol < 0.002) { EngineCurrentVol = 0 }
        if (BoostSoundPlaying & (abs(EngineCurrentVol - EngineVolLast) > 0.002)) {
            soundVolume(30, EngineCurrentVol)
            soundPitch(30, clamp(45 + (EngineCurrentVol/max(EngineVolume, 0.01))*45, 30, 130))
            EngineVolLast = EngineCurrentVol
        }
        if (!BoostAny & BoostSoundPlaying & EngineCurrentVol == 0) { soundStop(30) BoostSoundPlaying = 0 EngineVolLast = -1 }
        BoostWasOn = BoostAny

        if (!SlideReady) {
            CurBase:soundPlay(31, 0, "SolidMetal.ScrapeSmooth")
            soundPitch(31, SlidePitchMin)
            soundVolume(31, 0)
            SlideReady = 1
        }
        if (Active & !Edit & On_Ground & (GroundSpeed2 > 30)) { SlideTargetVol = clamp((GroundSpeed2 - 30)/SpeedRef, 0, 1)*SlideVolMax }
        else { SlideTargetVol = 0 }
        SlideVol = SlideVol + (SlideTargetVol - SlideVol)*0.15
        if (SlideVol < 0.002) { SlideVol = 0 }
        if (SlideReady) {
            local SPitch = clamp(SlidePitchMin + (GroundSpeed2/SpeedRef)*(SlidePitchMax - SlidePitchMin), SlidePitchMin, SlidePitchMax)
            if ((abs(SlideVol - SlideVolLast) > 0.002) | (abs(SPitch - SlidePitchLast) > 0.5)) {
                soundVolume(31, SlideVol)
                soundPitch(31, SPitch)
                SlideVolLast = SlideVol
                SlidePitchLast = SPitch
            }
        }
    }
}

if (changed(~ZMIn) & ZMIn) {
    if (ThirdPerson) { TP_DistTarget = clamp(TP_DistTarget - TP_DistStep, TP_DistMin, TP_DistMax) }
    else { if (ZMState < FOV:count()-1) { ZMState++ } }
}
if (changed(~ZMOut) & ZMOut) {
    if (ThirdPerson) { TP_DistTarget = clamp(TP_DistTarget + TP_DistStep, TP_DistMin, TP_DistMax) }
    else { if (ZMState > 0) { ZMState-- } }
}
if (changed(Active) & Active) {
    Idle = 0
    CurBase:propSetVelocity(vec(0))
    if (CurBase:isFrozen()) { CurBase:propFreeze(0) }
}
